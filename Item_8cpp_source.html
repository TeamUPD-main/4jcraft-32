1494HitResult* Item::getPlayerPOVHitResult(Level* level,
1495 std::shared_ptr<Player> player,
1496 bool alsoPickLiquid) {
1497 float a = 1;
1498
1499 float xRot = player->xRotO + (player->xRot - player->xRotO) * a;
1500 float yRot = player->yRotO + (player->yRot - player->yRotO) * a;
1501
1502 double x = player->xo + (player->x - player->xo) * a;
1503 double y =
1504 player->yo + (player->y - player->yo) * a + 1.62 - player->heightOffset;
1505 double z = player->zo + (player->z - player->zo) * a;
1506
1507 Vec3* from = Vec3::newTemp(x, y, z);
1508
1509 float yCos = (float)cos(-yRot * Mth::RAD_TO_GRAD - PI);
1510 float ySin = (float)sin(-yRot * Mth::RAD_TO_GRAD - PI);
1511 float xCos = (float)-cos(-xRot * Mth::RAD_TO_GRAD);
1512 float xSin = (float)sin(-xRot * Mth::RAD_TO_GRAD);
1513
1514 float xa = ySin * xCos;
1515 float ya = xSin;
1516 float za = yCos * xCos;
1517
1518 double range = 5;
1519 Vec3* to = from->add(xa * range, ya * range, za * range);
1520 return level->clip(from, to, alsoPickLiquid, !alsoPickLiquid);
1521}
