87bool DyePowderItem::useOn(std::shared_ptr<ItemInstance> itemInstance,
88 std::shared_ptr<Player> player, Level* level, int x,
89 int y, int z, int face, float clickX, float clickY,
90 float clickZ, bool bTestUseOnOnly) {
91 if (!player->mayBuild(x, y, z)) return false;
92
93 // 4J-PB - Adding a test only version to allow tooltips to be displayed
94 if (itemInstance->getAuxValue() == WHITE) {
95 // bone meal is a fertilizer, so instantly grow trees and stuff
96
97 int tile = level->getTile(x, y, z);
98 if (tile == Tile::sapling_Id) {
99 if (!bTestUseOnOnly) {
100 if (!level->isClientSide) {
101 ((Sapling*)Tile::sapling)
102 ->growTree(level, x, y, z, level->random);
103 itemInstance->count--;
104 }
105 }
106 return true;
107 } else if (tile == Tile::mushroom1_Id || tile == Tile::mushroom2_Id) {
108 if (!bTestUseOnOnly) {
109 if (!level->isClientSide) {
110 if (((Mushroom*)Tile::tiles[tile])
111 ->growTree(level, x, y, z, level->random)) {
112 itemInstance->count--;
113 }
114 }
115 }
116 return true;
117 } else if (tile == Tile::melonStem_Id || tile == Tile::pumpkinStem_Id) {
118 if (level->getData(x, y, z) == 7) return false;
119 if (!bTestUseOnOnly) {
120 if (!level->isClientSide) {
121 ((StemTile*)Tile::tiles[tile])
122 ->growCropsToMax(level, x, y, z);
123 itemInstance->count--;
124 }
125 }
126 return true;
127 } else if (tile == Tile::carrots_Id || tile == Tile::potatoes_Id) {
128 if (level->getData(x, y, z) == 7) return false;
129 if (!bTestUseOnOnly) {
130 if (!level->isClientSide) {
131 ((CropTile*)Tile::tiles[tile])
132 ->growCropsToMax(level, x, y, z);
133 itemInstance->count--;
134 }
135 }
136 return true;
137 } else if (tile == Tile::crops_Id) {
138 if (level->getData(x, y, z) == 7) return false;
139 if (!bTestUseOnOnly) {
140 if (!level->isClientSide) {
141 ((CropTile*)Tile::crops)->growCropsToMax(level, x, y, z);
142 itemInstance->count--;
143 }
144 }
145 return true;
146 } else if (tile == Tile::cocoa_Id) {
147 if (!bTestUseOnOnly) {
148 if (!level->isClientSide) {
149 level->setData(x, y, z,
150 (2 << 2) | DirectionalTile::getDirection(
151 level->getData(x, y, z)));
152 itemInstance->count--;
153 }
154 }
155 return true;
156 } else if (tile == Tile::grass_Id) {
157 if (!bTestUseOnOnly) {
158 if (!level->isClientSide) {
159 itemInstance->count--;
160
161 for (int j = 0; j < 128; j++) {
162 int xx = x;
163 int yy = y + 1;
164 int zz = z;
165 for (int i = 0; i < j / 16; i++) {
166 xx += random->nextInt(3) - 1;
167 yy += (random->nextInt(3) - 1) *
168 random->nextInt(3) / 2;
169 zz += random->nextInt(3) - 1;
170 if (level->getTile(xx, yy - 1, zz) !=
171 Tile::grass_Id ||
172 level->isSolidBlockingTile(xx, yy, zz)) {
173 goto mainloop;
174 }
175 }
176
177 if (level->getTile(xx, yy, zz) == 0) {
178 if (random->nextInt(10) != 0) {
179 if (Tile::tallgrass->canSurvive(level, xx, yy,
180 zz))
181 level->setTileAndData(
182 xx, yy, zz, Tile::tallgrass_Id,
183 TallGrass::TALL_GRASS);
184 } else if (random->nextInt(3) != 0) {
185 if (Tile::flower->canSurvive(level, xx, yy, zz))
186 level->setTile(xx, yy, zz, Tile::flower_Id);
187 } else {
188 if (Tile::rose->canSurvive(level, xx, yy, zz))
189 level->setTile(xx, yy, zz, Tile::rose_Id);
190 }
191 }
192
193 // 4J - Stops infinite loops.
194 mainloop:
195 continue;
196 }
197 }
198 }
199
200 return true;
201 }
202 } else if (itemInstance->getAuxValue() == BROWN) {
203 // plant cocoa
204
205 int tile = level->getTile(x, y, z);
206 int data = level->getData(x, y, z);
207
208 if (tile == Tile::treeTrunk_Id &&
209 TreeTile::getWoodType(data) == TreeTile::JUNGLE_TRUNK) {
210 if (face == 0) return false;
211 if (face == 1) return false;
212 if (face == 2) z--;
213 if (face == 3) z++;
214 if (face == 4) x--;
215 if (face == 5) x++;
216
217 if (!bTestUseOnOnly) {
218 if (level->isEmptyTile(x, y, z)) {
219 int cocoaData =
220 Tile::tiles[Tile::cocoa_Id]->getPlacedOnFaceDataValue(
221 level, x, y, z, face, clickX, clickY, clickZ, 0);
222 level->setTileAndData(x, y, z, Tile::cocoa_Id, cocoaData);
223 if (!player->abilities.instabuild) {
224 itemInstance->count--;
225 }
226 }
227 }
228 return true;
229 }
230 }
231 return false;
232}

234bool DyePowderItem::interactEnemy(std::shared_ptr<ItemInstance> itemInstance,
235 std::shared_ptr<Mob> mob) {
236 if (std::dynamic_pointer_cast<Sheep>(mob) != NULL) {
237 std::shared_ptr<Sheep> sheep = std::dynamic_pointer_cast<Sheep>(mob);
238 // convert to tile-based color value (0 is white instead of black)
239 int newColor =
240 ClothTile::getTileDataForItemAuxValue(itemInstance->getAuxValue());
241 if (!sheep->isSheared() && sheep->getColor() != newColor) {
242 sheep->setColor(newColor);
243 itemInstance->count--;
244 }
245 return true;
246 }
247 return false;
248}
