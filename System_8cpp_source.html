48// Returns the current value of the most precise available system timer, in
49// nanoseconds. This method can only be used to measure elapsed time and is not
50// related to any other notion of system or wall-clock time. The value returned
51// represents nanoseconds since some fixed but arbitrary time (perhaps in the
52// future, so values may be negative). This method provides nanosecond
53// precision, but not necessarily nanosecond accuracy. No guarantees are made
54// about how frequently values change. Differences in successive calls that span
55// greater than approximately 292 years (263 nanoseconds) will not accurately
56// compute elapsed time due to numerical overflow.
57//
58// For example, to measure how long some code takes to execute:
59//
60// long startTime = System.nanoTime();
61// // ... the code being measured ...
62// long estimatedTime = System.nanoTime() - startTime;
63//
64// Returns:
65// The current value of the system timer, in nanoseconds.
66__int64 System::nanoTime() {
67#if !defined(__linux__)
68 return GetTickCount() * 1000000LL;
69#else
70 struct timespec ts;
71 clock_gettime(CLOCK_MONOTONIC, &ts);
72 return static_cast<int64_t>(ts.tv_sec) * 1000000000LL + ts.tv_nsec;
73#endif // __linux__
74}

76// Returns the current time in milliseconds. Note that while the unit of time of
77// the return value is a millisecond, the granularity of the value depends on
78// the underlying operating system and may be larger. For example, many
79// operating systems measure time in units of tens of milliseconds. See the
80// description of the class Date for a discussion of slight discrepancies that
81// may arise between "computer time" and coordinated universal time (UTC).
82//
83// Returns:
84// the difference, measured in milliseconds, between the current time and
85// midnight, January 1, 1970 UTC.
86__int64 System::currentTimeMillis() {
87#ifdef __PS3__
88 // sys_time_get_current_time() obtains the elapsed time since Epoch
89 //(1970/01/01 00:00:00 UTC). The value is separated into two parts: sec
90 //stores the elapsed time in seconds, and nsec
91 // stores the value that is smaller than a second in nanoseconds.
92 sys_time_sec_t sec;
93 sys_time_nsec_t nsec;
94 sys_time_get_current_time(&sec, &nsec);
95 __int64 msec = (sec * 1000) + (nsec / (1000 * 1000));
96 return msec;
97
98#elif defined __ORBIS__
99 SceRtcTick tick;
100 int err = sceRtcGetCurrentTick(&tick);
101
102 return (int64_t)(tick.tick / 1000);
103#elif defined __PSVITA__
104 // AP - TRC states we can't use the RTC for measuring elapsed game time
105 return sceKernelGetProcessTimeWide() / 1000;
106/* SceDateTime Time;
107 sceRtcGetCurrentClockLocalTime(&Time);
108 __int64 systTime = (((((((Time.day * 24) + Time.hour) * 60) +
109 Time.minute) * 60) + Time.second) * 1000) + (Time.microsecond / 1000); return
110 systTime;*/
111#elif defined(__linux__)
112 struct timeval tv;
113 gettimeofday(&tv, NULL);
114 // Convert to milliseconds since unix epoch instead of windows file time
115 // time is expecting calculation to be between 10-30 ms.
116 return (int64_t)tv.tv_sec * 1000LL + tv.tv_usec / 1000;
117#else
118
119 SYSTEMTIME UTCSysTime;
120 GetSystemTime(&UTCSysTime);
121
122 // Represents as a 64-bit value the number of 100-nanosecond intervals since
123 // January 1, 1601
124 FILETIME UTCFileTime;

127 LARGE_INTEGER li;
128 li.HighPart = UTCFileTime.dwHighDateTime;
129 li.LowPart = UTCFileTime.dwLowDateTime;

132#endif // __PS3__
133}
