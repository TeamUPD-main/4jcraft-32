74void Creeper::tick() {
75 oldSwell = swell;
76 if (isAlive()) {
77 int swellDir = getSwellDir();
78 if (swellDir > 0 && swell == 0) {
79 level->playSound(shared_from_this(), eSoundType_RANDOM_FUSE, 1,
80 0.5f);
81 }
82 swell += swellDir;
83 if (swell < 0) swell = 0;
84 if (swell >= MAX_SWELL) {
85 swell = MAX_SWELL;
86 if (!level->isClientSide) {
87 bool destroyBlocks =
88 true; // level.getGameRules().getBoolean(GameRules.RULE_MOBGRIEFING);
89 if (isPowered())
90 level->explode(shared_from_this(), x, y, z, 6,
91 destroyBlocks);
92 else
93 level->explode(shared_from_this(), x, y, z, 3,
94 destroyBlocks);
95 remove();
96 }
97 }
98 }
99 Monster::tick();
100}
