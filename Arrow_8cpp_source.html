50Arrow::Arrow(Level* level, std::shared_ptr<Mob> mob,
51 std::shared_ptr<Mob> target, float power, float uncertainty)
52 : Entity(level) {
53 _init();
54
55 this->owner = mob;
56 if (std::dynamic_pointer_cast<Player>(mob) != NULL) pickup = PICKUP_ALLOWED;
57
58 y = mob->y + mob->getHeadHeight() - 0.1f;
59
60 double xd = target->x - mob->x;
61 double yd = (target->y + target->getHeadHeight() - 0.7f) - y;
62 double zd = target->z - mob->z;
63 double sd = sqrt(xd * xd + zd * zd);
64 if (sd < 0.0000001) return;
65
66 float yRot = (float)(atan2(zd, xd) * 180 / PI) - 90;
67 float xRot = (float)-(atan2(yd, sd) * 180 / PI);
68
69 double xdn = xd / sd;
70 double zdn = zd / sd;
71 moveTo(mob->x + xdn, y, mob->z + zdn, yRot, xRot);
72 heightOffset = 0;
73
74 float yo = (float)sd * 0.2f;
75 shoot(xd, yd + yo, zd, power, uncertainty);
76}

114void Arrow::shoot(double xd, double yd, double zd, float pow,
115 float uncertainty) {
116 float dist = (float)sqrt(xd * xd + yd * yd + zd * zd);
117
118 xd /= dist;
119 yd /= dist;
120 zd /= dist;
121
122 xd += (random->nextGaussian()) * 0.0075f * uncertainty;
123 yd += (random->nextGaussian()) * 0.0075f * uncertainty;
124 zd += (random->nextGaussian()) * 0.0075f * uncertainty;
125
126 xd *= pow;
127 yd *= pow;
128 zd *= pow;
129
130 this->xd = xd;
131 this->yd = yd;
132 this->zd = zd;
133
134 double sd = sqrt(xd * xd + zd * zd);
135
136 yRotO = this->yRot = (float)(atan2(xd, zd) * 180 / PI);
137 xRotO = this->xRot = (float)(atan2(yd, sd) * 180 / PI);
138 life = 0;
139}

388 yd *= inertia;
