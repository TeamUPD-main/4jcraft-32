45void ControlledByPlayerGoal::tick() {
46 std::shared_ptr<Player> player =
47 std::dynamic_pointer_cast<Player>(mob->rider.lock());
48 PathfinderMob* pig = (PathfinderMob*)mob;
49
50 float yrd = Mth::wrapDegrees(player->yRot - mob->yRot) * 0.5f;
51 if (yrd > 5) yrd = 5;
52 if (yrd < -5) yrd = -5;
53
54 mob->yRot = Mth::wrapDegrees(mob->yRot + yrd);
55 if (speed < maxSpeed) speed += (maxSpeed - speed) * 0.01f;
56 if (speed > maxSpeed) speed = maxSpeed;
57
58 int x = Mth::floor(mob->x);
59 int y = Mth::floor(mob->y);
60 int z = Mth::floor(mob->z);
61 float moveSpeed = speed;
62 if (boosting) {
63 if (boostTime++ > boostTimeTotal) {
64 boosting = false;
65 }
66 moveSpeed += moveSpeed * 1.15f *
67 Mth::sin((float)boostTime / boostTimeTotal * PI);
68 }
69
70 float friction = 0.91f;
71 if (mob->onGround) {
72 friction = 0.6f * 0.91f;
73 int t = mob->level->getTile(x, y, z);
74 if (t > 0) {
75 friction = Tile::tiles[t]->friction * 0.91f;
76 }
77 }
78 float friction2 = (0.6f * 0.6f * 0.91f * 0.91f * 0.6f * 0.91f) /
79 (friction * friction * friction);
80 float sin = Mth::sin(pig->yRot * PI / 180);
81 float cos = Mth::cos(pig->yRot * PI / 180);
82 float aproxSpeed = pig->getSpeed() * friction2;
83 float dist = std::max((int)moveSpeed, 1);
84 dist = aproxSpeed / dist;
85 float normMoveSpeed = moveSpeed * dist;
86 float xa = -(normMoveSpeed * sin);
87 float za = normMoveSpeed * cos;
88
89 if (Mth::abs(xa) > Mth::abs(za)) {
90 if (xa < 0) xa -= mob->bbWidth / 2.0f;
91 if (xa > 0) xa += mob->bbWidth / 2.0f;
92 za = 0;
93 } else {
94 xa = 0;
95 if (za < 0) za -= mob->bbWidth / 2.0f;
96 if (za > 0) za += mob->bbWidth / 2.0f;
97 }
98
99 int xt = Mth::floor(mob->x + xa);
100 int zt = Mth::floor(mob->z + za);
101
102 Node* size = new Node(Mth::floor(mob->bbWidth + 1),
103 Mth::floor(mob->bbHeight + player->bbHeight + 1),
104 Mth::floor(mob->bbWidth + 1));
105
106 if (x != xt || z != zt) {
107 if (PathFinder::isFree(mob, xt, y, zt, size, false, false, true) ==
108 PathFinder::TYPE_BLOCKED &&
109 PathFinder::isFree(mob, x, y + 1, z, size, false, false, true) ==
110 PathFinder::TYPE_OPEN &&
111 PathFinder::isFree(mob, xt, y + 1, zt, size, false, false, true) ==
112 PathFinder::TYPE_OPEN) {
113 pig->getJumpControl()->jump();
114 }
115 }
116
117 if (!player->abilities.instabuild && speed >= maxSpeed * 0.5f &&
118 mob->getRandom()->nextFloat() < 0.006f && !boosting) {
119 std::shared_ptr<ItemInstance> carriedItem = player->getCarriedItem();
120
121 if (carriedItem != NULL && carriedItem->id == Item::carrotOnAStick_Id) {
122 carriedItem->hurt(1, player);
123
124 if (carriedItem->count == 0) {
125 std::shared_ptr<ItemInstance> replacement =
126 std::shared_ptr<ItemInstance>(
127 new ItemInstance(Item::fishingRod));
128 replacement->setTag(carriedItem->tag);
129 player->inventory->items[player->inventory->selected] =
130 replacement;
131 }
132 }
133 }
134
135 mob->travel(0, moveSpeed);
136}
