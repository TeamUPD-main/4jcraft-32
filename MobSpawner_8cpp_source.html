423bool MobSpawner::isSpawnPositionOk(MobCategory* category, Level* level, int x,
424 int y, int z) {
425 // 4J - don't let this actually create/load a chunk that isn't here already
426 // - we'll let the normal updateDirtyChunks etc. processes do that, so it
427 // can happen on another thread
428 if (!level->hasChunkAt(x, y, z)) return false;
429
430#ifdef __PSVITA__
431 // AP - added this for Vita. Make sure a new spawn point has 2 chunks around
432 // it. This will make sure monsters don't keep getting spawned on the edge
433 // preventing other new monsters from being spawned
434 int r = 32;
435 if (!level->hasChunksAt(x - r, 0, z - r, x + r, 0, z + r)) {
436 return false;
437 }
438#endif
439
440 if (category->getSpawnPositionMaterial() == Material::water) {
441 // 4J - changed to spawn water things only in deep water
442 int yo = 0;
443 int liquidCount = 0;
444
445 while ((y - yo) >= 0 && (yo < 5)) {
446 if (level->getMaterial(x, y - yo, z)->isLiquid()) liquidCount++;
447 yo++;
448 }
449
450 // 4J - Sometimes deep water could be just a waterfall, so check that
451 // it's wide as well
452 bool inEnoughWater = false;
453 if (liquidCount == 5) {
454 if (level->getMaterial(x + 5, y, z)->isLiquid() &&
455 level->getMaterial(x - 5, y, z)->isLiquid() &&
456 level->getMaterial(x, y, z + 5)->isLiquid() &&
457 level->getMaterial(x, y, z - 5)->isLiquid()) {
458 inEnoughWater = true;
459 }
460 }
461
462 return inEnoughWater && !level->isSolidBlockingTile(x, y + 1, z);
463 } else {
464 if (!level->isTopSolidBlocking(x, y - 1, z)) return false;
465 int tt = level->getTile(x, y - 1, z);
466 return tt != Tile::unbreakable_Id &&
467 !level->isSolidBlockingTile(x, y, z) &&
468 !level->getMaterial(x, y, z)->isLiquid() &&
469 !level->isSolidBlockingTile(x, y + 1, z);
470 }
