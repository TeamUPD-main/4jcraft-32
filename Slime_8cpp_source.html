78void Slime::tick() {
79 if (!level->isClientSide && level->difficulty == Difficulty::PEACEFUL &&
80 getSize() > 0) {
81 removed = true;
82 }
83
84 squish = squish + (targetSquish - squish) * .5f;
85
86 oSquish = squish;
87 bool wasOnGround = this->onGround;
88 Mob::tick();
89 if (onGround && !wasOnGround) {
90 int size = getSize();
91 for (int i = 0; i < size * 8; i++) {
92 float dir = random->nextFloat() * PI * 2;
93 float d = random->nextFloat() * 0.5f + 0.5f;
94 float xd = Mth::sin(dir) * size * 0.5f * d;
95 float zd = Mth::cos(dir) * size * 0.5f * d;
96 level->addParticle(getParticleName(), x + xd, bb->y0, z + zd, 0, 0,
97 0);
98 }
99
100 if (doPlayLandSound()) {
101 level->playSound(
102 shared_from_this(), getSquishSound(), getSoundVolume(),
103 ((random->nextFloat() - random->nextFloat()) * 0.2f + 1.0f) /
104 0.8f);
105 }
106 targetSquish = -0.5f;
107 }
108 // 4J Stu - Brought forward from 1.3 in TU7 to fix lava slime render
109 else if (!onGround && wasOnGround) {
110 targetSquish = 1;
111 }
112 decreaseSquish();
113}

155void Slime::remove() {
156 int size = getSize();
157 if (!level->isClientSide && size > 1 && getHealth() <= 0) {
158 int count = 2 + random->nextInt(3);
159 for (int i = 0; i < count; i++) {
160 // The mob spawner can currently make a maximum of 25 slimes
161 // (limited to 50% of the total amount of monsters which is 50) and
162 // so limit to slightly more than this so we have some head room to
163 // make a few spawned children. Also always create at least one new
164 // slime since we are getting rid of this one anyway.
165 if (i == 0 || level->countInstanceOf(eTYPE_SLIME, true) < 35) {
166 float xd = (i % 2 - 0.5f) * size / 4.0f;
167 float zd = (i / 2 - 0.5f) * size / 4.0f;
168 std::shared_ptr<Slime> slime = createChild();
169 slime->setSize(size / 2);
170 slime->moveTo(x + xd, y + 0.5, z + zd,
171 random->nextFloat() * 360, 0);
172 level->addEntity(slime);
173 }
174 }
175 }
176 Mob::remove();
177}

179void Slime::playerTouch(std::shared_ptr<Player> player) {
180 if (isDealsDamage()) {
181 int size = getSize();
182 if (canSee(player) &&
183 this->distanceToSqr(player) < (0.6 * size) * (0.6 * size)) {
184 DamageSource* damageSource = DamageSource::mobAttack(
185 std::dynamic_pointer_cast<Mob>(shared_from_this()));
186 if (player->hurt(damageSource, getAttackDamage())) {
187 level->playSound(
188 shared_from_this(), eSoundType_MOB_SLIME_ATTACK, 1,
189 (random->nextFloat() - random->nextFloat()) * 0.2f + 1.0f);
190 }
191 delete damageSource;

193 }
