184void EntityRenderDispatcher::prepare(Level* level, Textures* textures,
185 Font* font, std::shared_ptr<Mob> player,
186 Options* options, float a) {
187 this->level = level;
188 this->textures = textures;
189 this->options = options;
190 this->cameraEntity = player;
191 this->font = font;
192
193 if (player->isSleeping()) {
194 int t = level->getTile(Mth::floor(player->x), Mth::floor(player->y),
195 Mth::floor(player->z));
196 if (t == Tile::bed_Id) {
197 int data =
198 level->getData(Mth::floor(player->x), Mth::floor(player->y),
199 Mth::floor(player->z));
200
201 int direction = data & 3;
202 playerRotY = (float)(direction * 90 + 180);
203 playerRotX = 0;
204 }
205 } else {
206 playerRotY = player->yRotO + (player->yRot - player->yRotO) * a;
207 playerRotX = player->xRotO + (player->xRot - player->xRotO) * a;
208 }
209
210 std::shared_ptr<Player> pl = std::dynamic_pointer_cast<Player>(player);
211 if (pl->ThirdPersonView() == 2) {
212 playerRotY += 180;
213 }
214
215 xPlayer = player->xOld + (player->x - player->xOld) * a;
216 yPlayer = player->yOld + (player->y - player->yOld) * a;
217 zPlayer = player->zOld + (player->z - player->zOld) * a;
218}

220void EntityRenderDispatcher::render(std::shared_ptr<Entity> entity, float a) {
221 double x = entity->xOld + (entity->x - entity->xOld) * a;
222 double y = entity->yOld + (entity->y - entity->yOld) * a;
223 double z = entity->zOld + (entity->z - entity->zOld) * a;
224
225 // Fix for #61057 - TU7: Gameplay: Boat is glitching when player float
226 // forward and turning. Fix to handle the case that yRot and yRotO wrap over
227 // the 0/360 line
228 float rotDiff = entity->yRot - entity->yRotO;
229 if (rotDiff > 180 || rotDiff < -180) {
230 if (entity->yRot > entity->yRotO) {
231 rotDiff = (entity->yRot - 360) - entity->yRotO;
232 } else {
233 rotDiff = entity->yRot - (entity->yRotO - 360);
234 }
235 }
236 float r = entity->yRotO + (rotDiff)*a;
237
238 int col = entity->getLightColor(a);
239 if (entity->isOnFire()) {
240 col = SharedConstants::FULLBRIGHT_LIGHTVALUE;
241 }
242 int u = col % 65536;
243 int v = col / 65536;
244 glMultiTexCoord2f(GL_TEXTURE1, u / 1.0f, v / 1.0f);
245 glColor4f(1, 1, 1, 1);
246
247 render(entity, x - xOff, y - yOff, z - zOff, r, a);
248}

262double EntityRenderDispatcher::distanceToSqr(double x, double y, double z) {
263 double xd = x - xPlayer;
264 double yd = y - yPlayer;
265 double zd = z - zPlayer;
266 return xd * xd + yd * yd + zd * zd;
267}
