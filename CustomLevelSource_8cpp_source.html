98void CustomLevelSource::prepareHeights(int xOffs, int zOffs, byteArray blocks) {
99#ifdef _OVERRIDE_HEIGHTMAP
100 int xChunks = 16 / CHUNK_WIDTH;
101 int yChunks = Level::maxBuildHeight / CHUNK_HEIGHT;
102 int waterHeight = level->seaLevel;
103
104 int xSize = xChunks + 1;
105 int ySize = Level::maxBuildHeight / CHUNK_HEIGHT + 1;
106 int zSize = xChunks + 1;
107
108 int xMapStart = xOffs + m_XZSize / 2;
109 int zMapStart = zOffs + m_XZSize / 2;
110 for (int xc = 0; xc < xChunks; xc++) {
111 for (int zc = 0; zc < xChunks; zc++) {
112 for (int yc = 0; yc < yChunks; yc++) {
113 for (int y = 0; y < CHUNK_HEIGHT; y++) {
114 for (int x = 0; x < CHUNK_WIDTH; x++) {
115 for (int z = 0; z < CHUNK_WIDTH; z++) {
116 int mapIndex =
117 (zMapStart * 16 + z + (zc * CHUNK_WIDTH)) *
118 (m_XZSize * 16) +
119 (xMapStart * 16 + x + (xc * CHUNK_WIDTH));
120 int mapHeight = m_heightmapOverride[mapIndex];
121 waterHeight = m_waterheightOverride[mapIndex];
122 // app.DebugPrintf("MapHeight = %d, y = %d\n",
123 // mapHeight, yc * CHUNK_HEIGHT + y);
125 // 4J - add this chunk of code to make land
126 // "fall-off" at the edges of a finite world - size
127 // of that world is currently hard-coded in here
128 const int worldSize = m_XZSize * 16;
129 const int falloffStart =
130 32; // chunks away from edge were we start
131 // doing fall-off
132 const float falloffMax =
133 128.0f; // max value we need to get to falloff
134 // by the edge of the map
135
136 int xxx = ((xOffs * 16) + x + (xc * CHUNK_WIDTH));
137 int zzz = ((zOffs * 16) + z + (zc * CHUNK_WIDTH));
138
139 // Get distance to edges of world in x
140 int xxx0 = xxx + (worldSize / 2);
141 if (xxx0 < 0) xxx0 = 0;
142 int xxx1 = ((worldSize / 2) - 1) - xxx;
143 if (xxx1 < 0) xxx1 = 0;
144
145 // Get distance to edges of world in z
146 int zzz0 = zzz + (worldSize / 2);
147 if (zzz0 < 0) zzz0 = 0;
148 int zzz1 = ((worldSize / 2) - 1) - zzz;
149 if (zzz1 < 0) zzz1 = 0;
150
151 // Get min distance to any edge
152 int emin = xxx0;
153 if (xxx1 < emin) emin = xxx1;
154 if (zzz0 < emin) emin = zzz0;
155 if (zzz1 < emin) emin = zzz1;
156
157 float comp = 0.0f;
158
159 // Calculate how much we want the world to fall
160 // away, if we're in the defined region to do so
161 if (emin < falloffStart) {
162 int falloff = falloffStart - emin;
163 comp = ((float)falloff / (float)falloffStart) *
164 falloffMax;
165 }
166 // 4J - end of extra code
168 int tileId = 0;
169 // 4J - this comparison used to just be with 0.0f
170 // but is now varied by block above
171 if (yc * CHUNK_HEIGHT + y < mapHeight) {
172 tileId = (std::uint8_t)Tile::rock_Id;
173 } else if (yc * CHUNK_HEIGHT + y < waterHeight) {
174 tileId = (std::uint8_t)Tile::calmWater_Id;
175 }
176
177 // 4J - more extra code to make sure that the column
178 // at the edge of the world is just water & rock, to
179 // match the infinite sea that continues on after
180 // the edge of the world.
181
182 if (emin == 0) {
183 // This matches code in MultiPlayerChunkCache
184 // that makes the geometry which continues at
185 // the edge of the world
186 if (yc * CHUNK_HEIGHT + y <=
187 (level->getSeaLevel() - 10))
188 tileId = Tile::rock_Id;
189 else if (yc * CHUNK_HEIGHT + y <
190 level->getSeaLevel())
191 tileId = Tile::calmWater_Id;
192 }
193
194 int indexY = (yc * CHUNK_HEIGHT + y);
195 int offsAdjustment = 0;
196 if (indexY >=
197 Level::COMPRESSED_CHUNK_SECTION_HEIGHT) {
198 indexY -=
199 Level::COMPRESSED_CHUNK_SECTION_HEIGHT;
200 offsAdjustment =
201 Level::COMPRESSED_CHUNK_SECTION_TILES;
202 }
203 int offs =
204 ((x + xc * CHUNK_WIDTH)
205 << Level::genDepthBitsPlusFour |
206 (z + zc * CHUNK_WIDTH) << Level::genDepthBits |
207 indexY) +
208 offsAdjustment;
209 blocks[offs] = tileId;
210 }
211 }
212 }
213 }
214 }

219void CustomLevelSource::buildSurfaces(int xOffs, int zOffs, byteArray blocks,
220 BiomeArray biomes) {
221#ifdef _OVERRIDE_HEIGHTMAP
222 int waterHeight = level->seaLevel;
223 int xMapStart = xOffs + m_XZSize / 2;
224 int zMapStart = zOffs + m_XZSize / 2;
225
226 double s = 1 / 32.0;
227
228 doubleArray depthBuffer(16 *
229 16); // 4J - used to be declared with class level
230 // scope but moved here for thread safety
231
232 depthBuffer = perlinNoise3->getRegion(depthBuffer, xOffs * 16, zOffs * 16,
233 0, 16, 16, 1, s * 2, s * 2, s * 2);
234
235 for (int x = 0; x < 16; x++) {
236 for (int z = 0; z < 16; z++) {
237 int mapIndex =
238 (zMapStart * 16 + z) * (m_XZSize * 16) + (xMapStart * 16 + x);
239 waterHeight = m_waterheightOverride[mapIndex];
240
241 Biome* b = biomes[z + x * 16];
242 float temp = b->getTemperature();
243 int runDepth = (int)(depthBuffer[x + z * 16] / 3 + 3 +
244 random->nextDouble() * 0.25);
245
246 int run = -1;
247
248 std::uint8_t top = b->topMaterial;
249 std::uint8_t material = b->material;
250
251 LevelGenerationOptions* lgo = app.getLevelGenerationOptions();
252 if (lgo != NULL) {
253 lgo->getBiomeOverride(b->id, material, top);
254 }
255
256 for (int y = Level::maxBuildHeight - 1; y >= 0; y--) {
257 int indexY = y;
258 int offsAdjustment = 0;
259 if (indexY >= Level::COMPRESSED_CHUNK_SECTION_HEIGHT) {
260 indexY -= Level::COMPRESSED_CHUNK_SECTION_HEIGHT;
261 offsAdjustment = Level::COMPRESSED_CHUNK_SECTION_TILES;
262 }
263 int offs = (x << Level::genDepthBitsPlusFour |
264 z << Level::genDepthBits | indexY) +
265 offsAdjustment;
266
267 if (y <= 1 + random->nextInt(
268 2)) // 4J - changed to make the bedrock not
269 // have bits you can get stuck in
270 // if (y <= 0 +
271 // random->nextInt(5))
272 {
273 blocks[offs] = (std::uint8_t)Tile::unbreakable_Id;
274 } else {
275 int old = blocks[offs];
276
277 if (old == 0) {
278 run = -1;
279 } else if (old == Tile::rock_Id) {
280 if (run == -1) {
281 if (runDepth <= 0) {
282 top = 0;
283 material = (std::uint8_t)Tile::rock_Id;
284 } else if (y >= waterHeight - 4 &&
285 y <= waterHeight + 1) {
286 top = b->topMaterial;
287 material = b->material;
288 if (lgo != NULL) {
289 lgo->getBiomeOverride(b->id, material, top);
290 }
291 }
292
293 if (y < waterHeight && top == 0) {
294 if (temp < 0.15f)
295 top = (std::uint8_t)Tile::ice_Id;
296 else
297 top = (std::uint8_t)Tile::calmWater_Id;
298 }
299
300 run = runDepth;
301 if (y >= waterHeight - 1)
302 blocks[offs] = top;
303 else
304 blocks[offs] = material;
305 } else if (run > 0) {
306 run--;
307 blocks[offs] = material;
308
309 // place a few sandstone blocks beneath sand
310 // runs
311 if (run == 0 && material == Tile::sand_Id) {
312 run = random->nextInt(4);
313 material = (std::uint8_t)Tile::sandStone_Id;
314 }
315 }
316 }
317 }
318 }
319 }
320 }

409void CustomLevelSource::calcWaterDepths(ChunkSource* parent, int xt, int zt) {
410#ifdef _OVERRIDE_HEIGHTMAP
411 int xo = xt * 16;
412 int zo = zt * 16;
413 for (int x = 0; x < 16; x++) {
414 int y = level->getSeaLevel();
415 for (int z = 0; z < 16; z++) {
416 int xp = xo + x + 7;
417 int zp = zo + z + 7;
418 int h = level->getHeightmap(xp, zp);
419 if (h <= 0) {
420 if (level->getHeightmap(xp - 1, zp) > 0 ||
421 level->getHeightmap(xp + 1, zp) > 0 ||
422 level->getHeightmap(xp, zp - 1) > 0 ||
423 level->getHeightmap(xp, zp + 1) > 0) {
424 bool hadWater = false;
425 if (hadWater ||
426 (level->getTile(xp - 1, y, zp) == Tile::calmWater_Id &&
427 level->getData(xp - 1, y, zp) < 7))
428 hadWater = true;
429 if (hadWater ||
430 (level->getTile(xp + 1, y, zp) == Tile::calmWater_Id &&
431 level->getData(xp + 1, y, zp) < 7))
432 hadWater = true;
433 if (hadWater ||
434 (level->getTile(xp, y, zp - 1) == Tile::calmWater_Id &&
435 level->getData(xp, y, zp - 1) < 7))
436 hadWater = true;
437 if (hadWater ||
438 (level->getTile(xp, y, zp + 1) == Tile::calmWater_Id &&
439 level->getData(xp, y, zp + 1) < 7))
440 hadWater = true;
441 if (hadWater) {
442 for (int x2 = -5; x2 <= 5; x2++) {
443 for (int z2 = -5; z2 <= 5; z2++) {
444 int d =
445 (x2 > 0 ? x2 : -x2) + (z2 > 0 ? z2 : -z2);
446
447 if (d <= 5) {
448 d = 6 - d;
449 if (level->getTile(xp + x2, y, zp + z2) ==
450 Tile::calmWater_Id) {
451 int od =
452 level->getData(xp + x2, y, zp + z2);
453 if (od < 7 && od < d) {
454 level->setData(xp + x2, y, zp + z2,
455 d);
456 }
457 }
458 }
459 }
460 }
461 if (hadWater) {
462 level->setTileAndDataNoUpdate(
463 xp, y, zp, Tile::calmWater_Id, 7);
464 for (int y2 = 0; y2 < y; y2++) {
465 level->setTileAndDataNoUpdate(
466 xp, y2, zp, Tile::calmWater_Id, 8);
467 }
468 }
469 }
470 }
471 }
472 }
