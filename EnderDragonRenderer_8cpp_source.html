100void EnderDragonRenderer::render(std::shared_ptr<Entity> _mob, double x,
101 double y, double z, float rot, float a) {
102 // 4J - dynamic cast required because we aren't using templates/generics in
103 // our version
104 std::shared_ptr<EnderDragon> mob =
105 std::dynamic_pointer_cast<EnderDragon>(_mob);
106 EnderDragonRenderer::bossInstance = mob;
107 if (currentModel != DragonModel::MODEL_ID) {
108 model = new DragonModel(0);
109 currentModel = DragonModel::MODEL_ID;
110 }
111 MobRenderer::render(mob, x, y, z, rot, a);
112 if (mob->nearestCrystal != NULL) {
113 float tt = mob->nearestCrystal->time + a;
114 float hh = sin(tt * 0.2f) / 2 + 0.5f;
115 hh = (hh * hh + hh) * 0.2f;
116
117 float xd = (float)(mob->nearestCrystal->x - mob->x -
118 (mob->xo - mob->x) * (1 - a));
119 float yd = (float)(hh + mob->nearestCrystal->y - 1 - mob->y -
120 (mob->yo - mob->y) * (1 - a));
121 float zd = (float)(mob->nearestCrystal->z - mob->z -
122 (mob->zo - mob->z) * (1 - a));
123
124 float sdd = sqrt(xd * xd + zd * zd);
125 float dd = sqrt(xd * xd + yd * yd + zd * zd);
126
127 // this fixes a problem when the dragon is hit and the beam goes black
128 // because the diffuse colour isn't being reset in MobRenderer::render
129 glColor4f(1, 1, 1, 1);
130
131 glPushMatrix();
132 glTranslatef((float)x, (float)y + 2, (float)z);
133 glRotatef((float)(-atan2(zd, xd)) * 180.0f / PI - 90.0f, 0, 1, 0);
134 glRotatef((float)(-atan2(sdd, yd)) * 180.0f / PI - 90.0f, 1, 0, 0);
135
136 // 4J-PB - Rotating the healing beam too
137 static float fRot = 0.0f;
138 glRotatef(fRot, 0, 0, 1);
139 fRot += 0.5f; // 4J - rate of rotation changed from 5.0 to 0.5 for
140 // photosensitivity reasons
141 if (fRot >= 360.0f) {
142 fRot = 0.0f;
143 }
144
145 Tesselator* t = Tesselator::getInstance();
146 Lighting::turnOff();
147 glDisable(GL_CULL_FACE);
148
149 glEnable(GL_BLEND);
150 glBlendFunc(GL_SRC_ALPHA, GL_DST_ALPHA);
151
152 bindTexture(
153 TN_MOB_ENDERDRAGON_BEAM); // 4J was "/mob/enderdragon/beam.png"
154
155 glShadeModel(GL_SMOOTH);
156
157 float v0 = 0 - (mob->tickCount + a) *
158 0.005f; // 4J - rate of movement changed from 0.01
159 // to 0.005 for photosensitivity reasons
160 float v1 = sqrt(xd * xd + yd * yd + zd * zd) / 32.0f -
161 (mob->tickCount + a) * 0.005f;
162
163 t->begin(GL_TRIANGLE_STRIP);
164
165 int steps = 8;
166 for (int i = 0; i <= steps; i++) {
167 double d = i % steps * PI * 2 / steps;
168 float s = sin(i % steps * PI * 2 / steps) * 0.75f;
169 float c = cos(i % steps * PI * 2 / steps) * 0.75f;
170 float u = i % steps * 1.0f / steps;
171 // t->color(0x000000);
172 t->vertexUV(s * 0.2f, c * 0.2f, 0, u, v1);
173 // t->color(0xffffff);
174 t->vertexUV(s, c, dd, u, v0);
175 }
176
177 t->end();
178 glEnable(GL_CULL_FACE);
179 glShadeModel(GL_FLAT);
180 glDisable(GL_BLEND);
181
182 glPopMatrix();
183 Lighting::turnOn();
184 }
