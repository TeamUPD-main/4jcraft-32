38void TheEndBiomeDecorator::decorate() {
39 decorateOres();
40
41 // this will only set the y to the top y of the chunks already processed...
42 int y = level->getTopSolidBlock(xo + 8, zo + 8);
43 if (y > level->GetHighestY()) level->SetHighestY(y);
44
45 // 4J-PB - editing to place 8 spikes in a circle, with increasing height
46
47 // are we within the chunk with a spike?
48 for (int i = 0; i < 8; i++) {
49 if ((xo == SpikeValA[i].iChunkX) && (zo == SpikeValA[i].iChunkZ)) {
50 // in the right chunk
51 spikeFeature->placeWithIndex(level, random, SpikeValA[i].x,
52 level->GetHighestY(), SpikeValA[i].z,
53 i, SpikeValA[i].radius);
54 }
55 }
56 if (xo == 0 && zo == 0) {
57 std::shared_ptr<EnderDragon> enderDragon =
58 std::shared_ptr<EnderDragon>(new EnderDragon(level));
59 enderDragon->moveTo(0, 128, 0, random->nextFloat() * 360, 0);
60 level->addEntity(enderDragon);
61 }
62
63 // end podium radius is 4, position is 0,0, so chunk needs to be the -16,-16
64 // one since this guarantees that all chunks required for the podium are
65 // loaded
66 if (xo == -16 && zo == -16) {
67 endPodiumFeature->place(level, random, 0, Level::genDepth / 2, 0);
68 }
