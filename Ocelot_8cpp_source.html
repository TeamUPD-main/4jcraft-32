164bool Ozelot::interact(std::shared_ptr<Player> player) {
165 std::shared_ptr<ItemInstance> item = player->inventory->getSelected();
166 if (isTame()) {
167 if (equalsIgnoreCase(player->getUUID(), getOwnerUUID())) {
168 if (!level->isClientSide && !isFood(item)) {
169 sitGoal->wantToSit(!isSitting());
170 }
171 }
172 } else {
173 if (temptGoal->isRunning() && item != NULL &&
174 item->id == Item::fish_raw_Id &&
175 player->distanceToSqr(shared_from_this()) < 3 * 3) {
176 // 4J-PB - don't lose the fish in creative mode
177 if (!player->abilities.instabuild) item->count--;
178 if (item->count <= 0) {
179 player->inventory->setItem(player->inventory->selected,
180 nullptr);
181 }
182
183 if (!level->isClientSide) {
184 if (random->nextInt(3) == 0) {
185 setTame(true);
186
187 // 4J-JEV, hook for durango event.
188 player->awardStat(
189 GenericStats::tamedEntity(eTYPE_OZELOT),
190 GenericStats::param_tamedEntity(eTYPE_OZELOT));
191
192 setCatType(1 + level->random->nextInt(3));
193 setOwnerUUID(player->getUUID());
194 spawnTamingParticles(true);
195 sitGoal->wantToSit(true);
196 level->broadcastEntityEvent(shared_from_this(),
197 EntityEvent::TAMING_SUCCEEDED);
198 } else {
199 spawnTamingParticles(false);
200 level->broadcastEntityEvent(shared_from_this(),
201 EntityEvent::TAMING_FAILED);
202 }
203 }
204 return true;
205 }
206 }
207 return TamableAnimal::interact(player);
208}

248bool Ozelot::canSpawn() {
249 // artificially make ozelots more rare
250 if (level->random->nextInt(3) == 0) {
251 return false;
252 }
253 if (level->isUnobstructed(bb) &&
254 level->getCubes(shared_from_this(), bb)->empty() &&
255 !level->containsAnyLiquid(bb)) {
256 int xt = Mth::floor(x);
257 int yt = Mth::floor(bb->y0);
258 int zt = Mth::floor(z);
259 if (yt < level->seaLevel) {
260 return false;
261 }
262
263 int tile = level->getTile(xt, yt - 1, zt);
264 if (tile == Tile::grass_Id || tile == Tile::leaves_Id) {
265 return true;
266 }
267 }
268 return false;
