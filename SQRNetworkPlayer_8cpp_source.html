206// Attempt to send data, of any size, from this player to that specified by pPlayerTarget. This may not be possible depending on the two players, due to
207// our star shaped network connectivity. Data may be any size, and is copied so on returning from this method it does not need to be preserved.
208void SQRNetworkPlayer::SendData( SQRNetworkPlayer *pPlayerTarget, const void *data, unsigned int dataSize )
209{
210 // Our network is connected as a star. If we are the host, then we can send to any remote player. If we're a client, we can send only to the host.
211 // The host can also send to other local players, but this doesn't need to go through Rudp.
212 if( m_host )
213 {
214 if( ( m_type == SNP_TYPE_HOST ) && ( pPlayerTarget->m_type == SNP_TYPE_LOCAL ) )
215 {
216 // Special internal communication from host to local player
217 m_manager->LocalDataSend( this, pPlayerTarget, data, dataSize );
218 }
219 else if( ( m_type == SNP_TYPE_LOCAL ) && ( pPlayerTarget->m_type == SNP_TYPE_HOST ) )
220 {
221 // Special internal communication from local player to host
222 m_manager->LocalDataSend( this, pPlayerTarget, data, dataSize );
223 }
224 else if( ( m_type == SNP_TYPE_HOST ) && ( pPlayerTarget->m_type == SNP_TYPE_REMOTE ) )
225 {
226 // Rudp communication from host to remote player - handled by remote player instance
227 pPlayerTarget->SendInternal(data,dataSize);
228 }
229 else
230 {
231 // Can't do any other types of communications
232 assert(false);
233 }
234 }
235 else
236 {
237 if( ( m_type == SNP_TYPE_LOCAL ) && ( pPlayerTarget->m_type == SNP_TYPE_HOST ) )
238 {
239 // Rudp communication from client to host - handled by this player instace
240 SendInternal(data, dataSize);
241 }
242 else
243 {
244 // Can't do any other types of communications
245 assert(false);
246 }
247 }
248}

250// Internal send function - to simplify the number of mechanisms we have for sending data, this method just adds the data to be send to the player's internal queue,
251// and then calls SendMoreInternal. This method can take any size of data, which it will split up into payload size chunks before sending. All input data is copied
252// into internal buffers.
253void SQRNetworkPlayer::SendInternal(const void *data, unsigned int dataSize)
254{
255 EnterCriticalSection(&m_csQueue);
256
257 QueuedSendBlock sendBlock;
258
259 unsigned char *dataCurrent = (unsigned char *)data;
260 unsigned int dataRemaining = dataSize;
261
262 while( dataRemaining )
263 {
264 int dataSize = dataRemaining;
265 if( dataSize > SNP_MAX_PAYLOAD ) dataSize = SNP_MAX_PAYLOAD;
266 sendBlock.start = new unsigned char [dataSize];
267 sendBlock.end = sendBlock.start + dataSize;
268 sendBlock.current = sendBlock.start;
269 memcpy( sendBlock.start, dataCurrent, dataSize);
270 m_sendQueue.push(sendBlock);
271 dataRemaining -= dataSize;
272 dataCurrent += dataSize;
273 }
274
275 // Now try and send as much as we can
276 SendMoreInternal();
277
278 LeaveCriticalSection(&m_csQueue);
279}

284// Internal send function. This attempts to send as many elements in the queue as possible until the write function tells us that we can't send any more. This way,
285// we are guaranteed that if there *is* anything more in the queue left to send, we'll get a CELL_RUDP_CONTEXT_EVENT_WRITABLE event when whatever we've managed to
286// send here is complete, and can continue on.
287void SQRNetworkPlayer::SendMoreInternal()
288{
289 EnterCriticalSection(&m_csQueue);
290 bool keepSending;
291 do
292 {
293 keepSending = false;
294 if( m_sendQueue.size() > 0)
295 {
296 // Attempt to send the full data in the first element in our queue
297 unsigned char *data= m_sendQueue.front().current;
298 int dataSize = m_sendQueue.front().end - m_sendQueue.front().current;
299#ifdef __PS3__
300 int ret = cellRudpWrite( m_rudpCtx, data, dataSize, 0);//CELL_RUDP_MSG_LATENCY_CRITICAL );
301 int wouldBlockFlag = CELL_RUDP_ERROR_WOULDBLOCK;
302
303#else // __ORBIS__
304 int ret = sceRudpWrite( m_rudpCtx, data, dataSize, 0);//CELL_RUDP_MSG_LATENCY_CRITICAL );
305 int wouldBlockFlag = SCE_RUDP_ERROR_WOULDBLOCK;
306#endif
307 if( ret == dataSize )
308 {
309 // Fully sent, remove from queue - will loop in the while loop to see if there's anything else in the queue we could send
310 delete [] m_sendQueue.front().start;
311 m_sendQueue.pop();
312 if( m_sendQueue.size() )
313 {
314 keepSending = true;
315 }
316 }
317 else if( ( ret >= 0 ) || ( ret == wouldBlockFlag ) )
318 {
319
320
321 // Things left to send - adjust this element in the queue
322 int remainingBytes;
323 if( ret >= 0 )
324 {
325 // Only ret bytes sent so far
326 remainingBytes = dataSize - ret;
327 assert(remainingBytes > 0 );
328 }
329 else
330 {
331 // Is CELL_RUDP_ERROR_WOULDBLOCK, nothing has yet been sent
332 remainingBytes = dataSize;
333 }
334 m_sendQueue.front().current = m_sendQueue.front().end - remainingBytes;
335 }
336 }
337 } while (keepSending);
338 LeaveCriticalSection(&m_csQueue);
339}
