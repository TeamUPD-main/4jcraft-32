67void FallingTile::tick() {
68 if (tile == 0) {
69 remove();

71 }
72
73 xo = x;
74 yo = y;
75 zo = z;
76 time++;
77
78 yd -= 0.04f;
79 move(xd, yd, zd);
80 xd *= 0.98f;
81 yd *= 0.98f;
82 zd *= 0.98f;
83
84 if (!level->isClientSide) {
85 int xt = Mth::floor(x);
86 int yt = Mth::floor(y);
87 int zt = Mth::floor(z);
88 if (time == 1) {
89 if (level->getTile(xt, yt, zt) == tile) {
90 level->setTile(xt, yt, zt, 0);
91 } else {
92 remove();
93 return;
94 }
95 }
96
97 if (onGround) {
98 xd *= 0.7f;
99 zd *= 0.7f;
100 yd *= -0.5f;
101
102 // if (HeavyTile.isFree(level, xt, yt, zt)) {
103 if (level->getTile(xt, yt, zt) != Tile::pistonMovingPiece_Id) {
104 remove();
105 if (!cancelDrop &&
106 level->mayPlace(tile, xt, yt, zt, true, 1, nullptr) &&
107 !HeavyTile::isFree(level, xt, yt - 1, zt) &&
108 level->setTileAndData(xt, yt, zt, tile, data)) {
109 HeavyTile* hv = dynamic_cast<HeavyTile*>(Tile::tiles[tile]);
110 if (hv) {
111 hv->onLand(level, xt, yt, zt, data);
112 }
113 } else {
114 if (dropItem && !cancelDrop)
115 spawnAtLocation(
116 std::shared_ptr<ItemInstance>(new ItemInstance(
117 tile, 1,
118 Tile::tiles[tile]->getSpawnResourcesAuxValue(
119 data))),
120 0);
121 }
122 }
123 } else if ((time > 20 * 5 && !level->isClientSide &&
124 (yt < 1 || yt > Level::maxBuildHeight)) ||
125 (time > 20 * 30)) {
126 if (dropItem) spawnAtLocation(tile, 1);
127 remove();
128 }
129 }
