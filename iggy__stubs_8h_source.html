63 fp->props.movie_frame_rate_current_in_fps = 30.0f;
64 fprintf(stderr, "[Iggy Stub] Created fake player %d (data=%p, size=%u)\n",
65 s_fakePlayerCount - 1, data, data_size_in_bytes);
