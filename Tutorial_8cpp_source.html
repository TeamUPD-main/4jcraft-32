1154void Tutorial::tick()
1155{
1156 // Don't do anything for the first 2 seconds so that the loading screen is gone
1157 if(!m_bHasTickedOnce)
1158 {
1159 int time = GetTickCount();
1160 if(m_firstTickTime == 0)
1161 {
1162 m_firstTickTime = time;
1163 }
1164 else if ( time - m_firstTickTime > 1500 )
1165 {
1166 m_bHasTickedOnce = true;
1167 }
1168 }
1169 if(!m_bHasTickedOnce)
1170 {
1171 return;
1172 }
1173
1174 bool constraintChanged = false;
1175 bool taskChanged = false;
1176
1177 for(unsigned int state = 0; state < e_Tutorial_State_Max; ++state)
1178 {
1179 AUTO_VAR(it, constraintsToRemove[state].begin());
1180 while(it < constraintsToRemove[state].end() )
1181 {
1182 ++(*it).second;
1183 if( (*it).second > m_iTutorialConstraintDelayRemoveTicks )
1184 {
1185 TutorialConstraint *c = (*it).first;
1186 constraints[state].erase( find( constraints[state].begin(), constraints[state].end(), c) );
1187 c->setQueuedForRemoval(false);
1188 it = constraintsToRemove[state].erase( it );
1189
1190 if( c->getDeleteOnDeactivate() )
1191 {
1192 delete c;
1193 }
1194 }
1195 else
1196 {
1197 ++it;
1198 }
1199 }
1200 }
1201
1202 // 4J Stu TODO - Make this a constraint
1203 Minecraft *pMinecraft = Minecraft::GetInstance();
1204 if(m_freezeTime && !m_timeFrozen && !m_fullTutorialComplete )
1205 {
1206 // Need to set the time on both levels to stop the flickering as the local level
1207 // tries to predict the time
1208 MinecraftServer::SetTimeOfDay(m_iTutorialFreezeTimeValue);
1209 pMinecraft->level->setOverrideTimeOfDay(m_iTutorialFreezeTimeValue); // Always daytime
1210 m_timeFrozen = true;
1211 }
1212 else if(m_freezeTime && m_timeFrozen && m_fullTutorialComplete)
1213 {
1214 __int64 currentTime = pMinecraft->level->getTime();
1215 int currentDayTime = (currentTime % Level::TICKS_PER_DAY);
1216 int timeToAdd = 0;
1217 if(currentDayTime > m_iTutorialFreezeTimeValue)
1218 {
1219 timeToAdd = (Level::TICKS_PER_DAY - currentDayTime) + m_iTutorialFreezeTimeValue;
1220 }
1221 else
1222 {
1223 timeToAdd = m_iTutorialFreezeTimeValue - currentDayTime;
1224 }
1225 __int64 targetTime = currentTime + timeToAdd;
1226 MinecraftServer::SetTimeOfDay(-1);
1227 MinecraftServer::SetTime(targetTime);
1228 pMinecraft->level->setOverrideTimeOfDay(-1);
1229 pMinecraft->level->setTime(targetTime);
1230 m_timeFrozen = false;
1231 }
1232
1233 if(!m_allowShow)
1234 {
1235 if( currentTask[m_CurrentState] != NULL && (!currentTask[m_CurrentState]->AllowFade() || (lastMessageTime + m_iTutorialDisplayMessageTime ) > GetTickCount() ) )
1236 {
1237 uiTempDisabled = true;
1238 }
1239 ui.SetTutorialVisible( m_iPad, false );
1240 return;
1241 }
1242
1243
1244 if(!hasRequestedUI )
1245 {
1246#ifdef _XBOX
1247 m_bSceneIsSplitscreen=app.GetLocalPlayerCount()>1;
1248 if(m_bSceneIsSplitscreen)
1249 {
1250 app.NavigateToScene(m_iPad, eUIComponent_TutorialPopup,(void *)this, false, false, &m_hTutorialScene);
1251 }
1252 else
1253 {
1254 app.NavigateToScene(m_iPad, eUIComponent_TutorialPopup,(void *)this, false, false, &m_hTutorialScene);
1255 }
1256#else
1257 ui.SetTutorial(m_iPad, this);
1258#endif
1259 hasRequestedUI = true;
1260 }
1261 else
1262 {
1263 // if we've changed mode, we may need to change scene
1264 if(m_bSceneIsSplitscreen != (app.GetLocalPlayerCount() > 1))
1265 {
1266#ifdef _XBOX
1267 app.TutorialSceneNavigateBack(m_iPad);
1268 m_bSceneIsSplitscreen=app.GetLocalPlayerCount()>1;
1269 if(m_bSceneIsSplitscreen)
1270 {
1271 app.NavigateToScene(m_iPad, eUIComponent_TutorialPopup,(void *)this, false, false, &m_hTutorialScene);
1272 }
1273 else
1274 {
1275 app.NavigateToScene(m_iPad, eUIComponent_TutorialPopup,(void *)this, false, false, &m_hTutorialScene);
1276 }
1277#else
1278 ui.SetTutorial(m_iPad, this);
1279#endif
1280 }
1281 }
1282
1283 if(ui.IsPauseMenuDisplayed( m_iPad ) )
1284 {
1285 if( currentTask[m_CurrentState] != NULL && (!currentTask[m_CurrentState]->AllowFade() || (lastMessageTime + m_iTutorialDisplayMessageTime ) > GetTickCount() ) )
1286 {
1287 uiTempDisabled = true;
1288 }
1289 ui.SetTutorialVisible( m_iPad, false );
1290 return;
1291 }
1292 if( uiTempDisabled )
1293 {
1294 ui.SetTutorialVisible( m_iPad, true );
1295 lastMessageTime = GetTickCount();
1296 uiTempDisabled = false;
1297 }
1298
1299 // Check constraints
1300 for(AUTO_VAR(it, m_globalConstraints.begin()); it < m_globalConstraints.end(); ++it)
1301 {
1302 TutorialConstraint *constraint = *it;
1303 constraint->tick(m_iPad);
1304 }
1305
1306 // Check hints
1307 int hintNeeded = -1;
1308 if(!m_hintDisplayed)
1309 {
1310 // 4J Stu - TU-1 interim
1311 // Allow turning off all the hints
1312 bool hintsOn = m_isFullTutorial || app.GetGameSettings(m_iPad,eGameSetting_Hints);
1313
1314 if(hintsOn)
1315 {
1316 for(AUTO_VAR(it, hints[m_CurrentState].begin()); it < hints[m_CurrentState].end(); ++it)
1317 {
1318 TutorialHint *hint = *it;
1319 hintNeeded = hint->tick();
1320 if(hintNeeded >= 0)
1321 {
1322 PopupMessageDetails *message = new PopupMessageDetails();
1323 message->m_messageId = hintNeeded;
1324 message->m_allowFade = hint->allowFade();
1325 message->m_forceDisplay = true;
1326 setMessage( hint, message );

1328 }
1329 }
1330 }
1331 }

1333 // Check constraints
1334 // Only need to update these if we aren't already failing something
1335 if( !m_allTutorialsComplete && (currentFailedConstraint[m_CurrentState] == NULL || currentFailedConstraint[m_CurrentState]->isConstraintSatisfied(m_iPad)) )
1336 {
1337 if( currentFailedConstraint[m_CurrentState] != NULL && currentFailedConstraint[m_CurrentState]->isConstraintSatisfied(m_iPad) )
1338 {
1339 constraintChanged = true;
1340 currentFailedConstraint[m_CurrentState] = NULL;

1343 {
1344 TutorialConstraint *constraint = *it;
1345 if( !constraint->isConstraintSatisfied(m_iPad) && constraint->isConstraintRestrictive(m_iPad) )
1346 {
1347 constraintChanged = true;
1348 currentFailedConstraint[m_CurrentState] = constraint;
1349 }
1350 }
1351 }
1352
1353 if( !m_allTutorialsComplete && currentFailedConstraint[m_CurrentState] == NULL )
1354 {
1355 // Update tasks
1356 bool isCurrentTask = true;
1357 AUTO_VAR(it, activeTasks[m_CurrentState].begin());
1358 while(activeTasks[m_CurrentState].size() > 0 && it < activeTasks[m_CurrentState].end())
1359 {
1360 TutorialTask *task = *it;
1361 if( isCurrentTask || task->isPreCompletionEnabled() )
1362 {
1363 isCurrentTask = false;
1364 if(
1365 ( !task->ShowMinimumTime() || ( task->hasBeenActivated() && (lastMessageTime + m_iTutorialMinimumDisplayMessageTime ) < GetTickCount() ) )
1366 && task->isCompleted()
1367 )
1368 {
1369 eTutorial_CompletionAction compAction = task->getCompletionAction();
1370 it = activeTasks[m_CurrentState].erase( it );
1371 delete task;
1372 task = NULL;
1373
1374 if( activeTasks[m_CurrentState].size() > 0 )
1375 {
1376 switch( compAction )
1377 {
1378 case e_Tutorial_Completion_Complete_State_Gameplay_Constraints:
1379 {
1380 // 4J Stu - Move the delayed constraints to the gameplay state so that they are in
1381 // effect for a bit longer
1382 AUTO_VAR(itCon, constraintsToRemove[m_CurrentState].begin());
1383 while(itCon != constraintsToRemove[m_CurrentState].end() )
1384 {
1385 constraints[e_Tutorial_State_Gameplay].push_back(itCon->first);
1386 constraintsToRemove[e_Tutorial_State_Gameplay].push_back( std::pair<TutorialConstraint *, unsigned char>(itCon->first, itCon->second) );
1387
1388 constraints[m_CurrentState].erase( find( constraints[m_CurrentState].begin(), constraints[m_CurrentState].end(), itCon->first) );
1389 itCon = constraintsToRemove[m_CurrentState].erase(itCon);
1390 }
1391 }
1392 // Fall through the the normal complete state
1393 case e_Tutorial_Completion_Complete_State:
1394 for(AUTO_VAR(itRem, activeTasks[m_CurrentState].begin()); itRem < activeTasks[m_CurrentState].end(); ++itRem)
1395 {
1396 delete (*itRem);
1397 }
1398 activeTasks[m_CurrentState].clear();
1399 break;
1400 case e_Tutorial_Completion_Jump_To_Last_Task:
1401 {
1402 TutorialTask *lastTask = activeTasks[m_CurrentState].at( activeTasks[m_CurrentState].size() - 1 );
1403 activeTasks[m_CurrentState].pop_back();
1404 for(AUTO_VAR(itRem, activeTasks[m_CurrentState].begin()); itRem < activeTasks[m_CurrentState].end(); ++itRem)
1405 {
1406 delete (*itRem);
1407 }
1408 activeTasks[m_CurrentState].clear();
1409 activeTasks[m_CurrentState].push_back( lastTask );
1410 it = activeTasks[m_CurrentState].begin();
1411 }
1412 break;
1413 case e_Tutorial_Completion_None:
1414 default:
1415 break;
1416 }
1417 }
1418
1419 if( activeTasks[m_CurrentState].size() > 0 )
1420 {
1421 currentTask[m_CurrentState] = activeTasks[m_CurrentState][0];
1422 currentTask[m_CurrentState]->setAsCurrentTask();
1423 }
1424 else
1425 {
1426 setStateCompleted( m_CurrentState );
1427
1428 currentTask[m_CurrentState] = NULL;
1429 }
1430 taskChanged = true;
1431
1432 // If we can complete this early, check if we can complete it right now
1433 if( currentTask[m_CurrentState] != NULL && currentTask[m_CurrentState]->isPreCompletionEnabled() )
1434 {
1435 isCurrentTask = true;
1436 }
1437 }
1438 else
1439 {
1440 ++it;
1441 }
