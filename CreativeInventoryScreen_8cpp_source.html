49std::shared_ptr<ItemInstance> CreativeInventoryScreen::ContainerCreative::clicked(int slotIndex, int buttonNum, int clickType, std::shared_ptr<Player> player)
50{
51 std::shared_ptr<Inventory> inventory = player->inventory;
52 std::shared_ptr<ItemInstance> carried = inventory->getCarried();
53
54 // Handle clicks outside the GUI
55 if (slotIndex == CLICKED_OUTSIDE)
56 {
57 // Drop the carried item
58 if (carried != NULL)
59 {
60 if (buttonNum == 0)
61 {
62 player->drop(carried, true);
63 inventory->setCarried(std::shared_ptr<ItemInstance>());
64 }
65 else if (buttonNum == 1)
66 {
67 std::shared_ptr<ItemInstance> single = carried->copy();
68 single->count = 1;
69 player->drop(single, true);
70 carried->count--;
71 if (carried->count <= 0)
72 {
73 inventory->setCarried(std::shared_ptr<ItemInstance>());
74 }
75 }
76 }
77 return std::shared_ptr<ItemInstance>();
78 }
79
80 // Validate slot index
81 if (slotIndex < 0 || slotIndex >= (int)slots->size())
82 {
83 return std::shared_ptr<ItemInstance>();
84 }
85
86 Slot* slot = slots->at(slotIndex);
87
88 // Handle creative inventory slots (0-44)
89 if (slotIndex >= 0 && slotIndex < ITEMS_PER_PAGE)
90 {
91 std::shared_ptr<ItemInstance> slotItem = slot->getItem();
92
93 // Handle SWAP (number key) - copy item to hotbar
94 if (clickType == CLICK_SWAP)
95 {
96 if (slotItem != NULL && buttonNum >= 0 && buttonNum < 9)
97 {
98 std::shared_ptr<ItemInstance> copy = slotItem->copy();
99 copy->count = copy->getMaxStackSize();
100 inventory->setItem(buttonNum, copy);
101 }
102 return std::shared_ptr<ItemInstance>();
103 }
104
105 // Handle CLONE (middle click)
106 if (clickType == CLICK_CLONE)
107 {
108 if (slotItem != NULL)
109 {
110 std::shared_ptr<ItemInstance> copy = slotItem->copy();
111 copy->count = copy->getMaxStackSize();
112 inventory->setCarried(copy);
113 }
114 return std::shared_ptr<ItemInstance>();
115 }
116
117 // Handle normal clicks
118 if (slotItem != NULL)
119 {
120 if (buttonNum == 0) // Left click
121 {
122 std::shared_ptr<ItemInstance> copy = slotItem->copy();
123 copy->count = copy->getMaxStackSize();
124 inventory->setCarried(copy);
125 }
126 else if (buttonNum == 1) // Right click
127 {
128 std::shared_ptr<ItemInstance> copy = slotItem->copy();
129 copy->count = 1;
130 inventory->setCarried(copy);
131 }
132 }
133 else if (carried != NULL)
134 {
135 // Clicking on empty creative slot with item - clear the carried item
136 inventory->setCarried(std::shared_ptr<ItemInstance>());
137 }
138
139 return std::shared_ptr<ItemInstance>();
140 }
141
142 // For hotbar slots (45-53), use normal container behavior
143 return AbstractContainerMenu::clicked(slotIndex, buttonNum, clickType, player);
144}

208void CreativeInventoryScreen::updateEvents()
209{
210#ifdef ENABLE_JAVA_GUIS
211 // Handle mouse wheel scrolling.
212 // We use ButtonDown with the scroll actions rather than GetScrollDelta() because
213 // both share s_scrollTicksForButtonPressed; whichever is called first in a tick
214 // zeroes it, so GetScrollDelta() would return 0 if hotbar scroll ran first.
215 // ButtonDown/ScrollSnap() snapshots once per tick so all callers see the same value.
216 if (needsScrollBars())
217 {
218 ContainerCreative* container = (ContainerCreative*)menu;
219 int totalRows = ((int)container->itemList.size() + COLUMNS - 1) / COLUMNS;
220 int scrollableRows = totalRows - ROWS;
221 if (scrollableRows > 0)
222 {
223 float step = 1.0f / (float)scrollableRows;
224 if (InputManager.ButtonDown(0, MINECRAFT_ACTION_LEFT_SCROLL))
225 {
226 currentScroll -= step;
227 currentScroll = std::max(0.0f, std::min(1.0f, currentScroll));
228 container->scrollTo(currentScroll);
229 }
230 else if (InputManager.ButtonDown(0, MINECRAFT_ACTION_RIGHT_SCROLL))
231 {
232 currentScroll += step;
233 currentScroll = std::max(0.0f, std::min(1.0f, currentScroll));
234 container->scrollTo(currentScroll);
235 }
236 }
237 }
238#endif
239 Screen::updateEvents();
240}

256void CreativeInventoryScreen::mouseClicked(int x, int y, int buttonNum)
257{
258 if (buttonNum == 0) isLeftMouseDown = true;
259
260 Screen::mouseClicked(x, y, buttonNum);
261
262 if (buttonNum == 0 || buttonNum == 1)
263 {
264 int i = x - (width - imageWidth) / 2;
265 int j = y - (height - imageHeight) / 2;
266
267 // Check for tab clicks first; let mouseReleased handle the actual tab switch
268 for (int tab = 0; tab < IUIScene_CreativeMenu::eCreativeInventoryTab_COUNT; ++tab)
269 {
270 if (isMouseOverTab(tab, i, j))
271 {
272 return;
273 }
274 }
275
276 // Determine which slot (if any) was clicked
277 Slot *slot = findSlot(x, y);
278
279 int xo = (width - imageWidth) / 2;
280 int yo = (height - imageHeight) / 2;
281 bool clickedOutside = (x < xo || y < yo || x >= xo + imageWidth || y >= yo + imageHeight);
282
283 int slotId = -1;
284 if (slot != NULL) slotId = slot->index;
285 if (clickedOutside) slotId = AbstractContainerMenu::CLICKED_OUTSIDE;
286
287 if (slotId == -1) return;
288
289 bool quickKey = slotId != AbstractContainerMenu::CLICKED_OUTSIDE &&
290 (Keyboard::isKeyDown(Keyboard::KEY_LSHIFT) || Keyboard::isKeyDown(Keyboard::KEY_RSHIFT));
291 int clickType = quickKey ? AbstractContainerMenu::CLICK_QUICK_MOVE : AbstractContainerMenu::CLICK_PICKUP;
292
293 // 4jcraft: bypass AbstractContainerScreen::mouseClicked / handleInventoryMouseClick
294 // here intentionally. The normal path sends a ContainerClickPacket to the server,
295 // where player->containerMenu is still the InventoryMenu (45 slots). Creative slot
296 // indices 0-44 are valid in ContainerCreative but not in InventoryMenu, and hotbar
297 // indices 45-53 exceed InventoryMenu's slot count entirely, causing an out-of-range
298 // crash in AbstractContainerMenu::clicked on the server side.
299 // Instead we apply the click locally and sync hotbar changes via SetCreativeModeSlotPacket.
300 menu->clicked(slotId, buttonNum, clickType, minecraft->player);
301
302 // 4jcraft: sync hotbar slot changes to the server using SetCreativeModeSlotPacket.
303 // The packet handler (PlayerConnection::handleSetCreativeModeSlot) validates slots
304 // against InventoryMenu coordinates where the hotbar starts at USE_ROW_SLOT_START (36),
305 // so we must offset the local hotbar index (0-8) accordingly.
306 if (slotId >= ITEMS_PER_PAGE && slotId < ITEMS_PER_PAGE + 9)
307 {
308 int hotbarSlot = slotId - ITEMS_PER_PAGE;
309 std::shared_ptr<ItemInstance> hotbarItem = minecraft->player->inventory->getItem(hotbarSlot);
310 minecraft->gameMode->handleCreativeModeItemAdd(hotbarItem, hotbarSlot + InventoryMenu::USE_ROW_SLOT_START);
311 }
312 }
