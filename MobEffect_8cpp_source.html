162void MobEffect::applyEffectTick(std::shared_ptr<Mob> mob, int amplification) {
163 // Maybe move this to separate class implementations in the future?
164 if (id == regeneration->id) {
165 if (mob->getHealth() < mob->getMaxHealth()) {
166 mob->heal(1);
167 }
168 } else if (id == poison->id) {
169 if (mob->getHealth() > 1) {
170 mob->hurt(DamageSource::magic, 1);
171 }
172 } else if (id == hunger->id &&
173 std::dynamic_pointer_cast<Player>(mob) != NULL) {
174 // every tick, cause the same amount of exhaustion as when removing
175 // a block, times amplification
176 std::dynamic_pointer_cast<Player>(mob)->causeFoodExhaustion(
177 FoodConstants::EXHAUSTION_MINE * (amplification + 1));
178 } else if ((id == heal->id && !mob->isInvertedHealAndHarm()) ||
179 (id == harm->id && mob->isInvertedHealAndHarm())) {
180 mob->heal(6 << amplification);
181 } else if ((id == harm->id && !mob->isInvertedHealAndHarm()) ||
182 (id == heal->id && mob->isInvertedHealAndHarm())) {
183 mob->hurt(DamageSource::magic, 6 << amplification);
184 }
185}

187void MobEffect::applyInstantenousEffect(std::shared_ptr<Mob> source,
188 std::shared_ptr<Mob> mob,
189 int amplification, double scale) {
190 if ((id == heal->id && !mob->isInvertedHealAndHarm()) ||
191 (id == harm->id && mob->isInvertedHealAndHarm())) {
192 int amount = (int)(scale * (double)(6 << amplification) + .5);
193 mob->heal(amount);
194 } else if ((id == harm->id && !mob->isInvertedHealAndHarm()) ||
195 (id == heal->id && mob->isInvertedHealAndHarm())) {
196 int amount = (int)(scale * (double)(6 << amplification) + .5);
197 if (source == NULL) {
198 mob->hurt(DamageSource::magic, amount);
199 } else {
200 DamageSource* damageSource =
201 DamageSource::indirectMagic(mob, source);
202 mob->hurt(damageSource, amount);
203 delete damageSource;
204 }
205 }

220bool MobEffect::isDurationEffectTick(int remainingDuration, int amplification) {
221 // Maybe move this to separate class implementations in the future?
222 if (id == regeneration->id || id == poison->id) {
223 // tick intervals are 25, 12, 6..
224 int interval = 25 >> amplification;
225 if (interval > 0) {
226 return (remainingDuration % interval) == 0;
227 }
228 return true;
229 } else if (id == hunger->id) {
230 return true;
231 }
232
233 return false;

258std::wstring MobEffect::formatDuration(MobEffectInstance* instance) {
259 int duration = instance->getDuration();
260
261 int seconds = duration / SharedConstants::TICKS_PER_SECOND;
262 int minutes = seconds / 60;
263 seconds %= 60;
264
265 wchar_t temp[8];
266 ZeroMemory(&temp, 8 * (sizeof(wchar_t)));
267
268 if (seconds < 10) {

274 }
