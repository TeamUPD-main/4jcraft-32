41bool StructureFeature::postProcess(Level* level, Random* random, int chunkX,
42 int chunkZ) {
43 // 4J Stu - The x and z used to be offset by (+8) here, but that means we
44 // can miss out half structures on the edge of the world Normal feature
45 // generation offsets generation by half a chunk to ensure that it can
46 // generate the entire feature in chunks already created Structure features
47 // don't need this, as the PlaceBlock function only places blocks inside the
48 // BoundingBox specified, and parts of a struture piece can be added in more
49 // than one post-process call
50 int cx = ((unsigned)chunkX << 4); // + 8;
51 int cz = ((unsigned)chunkZ << 4); // + 8;
52
53 bool intersection = false;
54 for (AUTO_VAR(it, cachedStructures.begin()); it != cachedStructures.end();
55 it++) {
56 StructureStart* structureStart = it->second;
57
58 if (structureStart->isValid()) {
59 if (structureStart->getBoundingBox()->intersects(cx, cz, cx + 15,
60 cz + 15)) {
61 BoundingBox* bb = new BoundingBox(cx, cz, cx + 15, cz + 15);
62 structureStart->postProcess(level, random, bb);
63 delete bb;
64 intersection = true;
65 }
66 }
67 }
68
69 return intersection;
70}
