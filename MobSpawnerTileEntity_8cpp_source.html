45void MobSpawnerTileEntity::tick() {
46 if (!isNearPlayer()) {
47 return;
48 }
49
50 if (level->isClientSide) {
51 double xP = x + level->random->nextFloat();
52 double yP = y + level->random->nextFloat();
53 double zP = z + level->random->nextFloat();
54 level->addParticle(eParticleType_smoke, xP, yP, zP, 0, 0, 0);
55 level->addParticle(eParticleType_flame, xP, yP, zP, 0, 0, 0);
56
57 oSpin = spin;
58 spin += 1000 / 220.0f;
59 while (spin > 360) {
60 spin -= 360;
61 }
62 while (oSpin > 360) {
63 oSpin -= 360;
64 }
65 } else {
66 if (spawnDelay == -1) delay();
67
68 if (spawnDelay > 0) {
69 spawnDelay--;
70 return;
71 }
72
73 for (int c = 0; c < spawnCount; c++) {
74 std::shared_ptr<Mob> entity = std::dynamic_pointer_cast<Mob>(
75 EntityIO::newEntity(entityId, level));
76 if (entity == NULL) return;
77
78 Mob* entityPtr = entity.get();
79 std::vector<std::shared_ptr<Entity> >* vecNearby =
80 level->getEntitiesOfClass(
81 typeid(*entityPtr),
82 AABB::newTemp(x, y, z, x + 1, y + 1, z + 1)->grow(8, 4, 8));
83 int nearBy =
84 (int)vecNearby
85 ->size(); // 4J - IB, TODO, Mob contains no getClass
86 delete vecNearby;
87
88 if (nearBy >= 6) {
89 delay();
90 return;
91 }
92
93 // 4J added - our mobspawner tiles should only be spawning monsters.
94 // Also respect the global limits we have for those so we don't go
95 // creating silly numbers of them. Have set this limit slightly
96 // higher than the main spawner has so that this tile entity is more
97 // likely to actually make something (60 rather than 50)
98 if (level->countInstanceOf(eTYPE_MONSTER, false) >= 60) {
99 return;
100 }
101
102 if (entity != NULL) {
103 double xp = x + (level->random->nextDouble() -
104 level->random->nextDouble()) *
105 4;
106 double yp = y + level->random->nextInt(3) - 1;
107 double zp = z + (level->random->nextDouble() -
108 level->random->nextDouble()) *
109 4;
110 std::shared_ptr<Mob> mob =
111 std::dynamic_pointer_cast<Mob>(entity);
112
113 entity->moveTo(xp, yp, zp, level->random->nextFloat() * 360, 0);
114
115 if (mob == NULL || mob->canSpawn()) {
116 fillExtraData(entity);
117
118 level->addEntity(entity);
119
120 level->levelEvent(LevelEvent::PARTICLES_MOBTILE_SPAWN, x, y,
121 z, 0);
122
123 if (mob != NULL) mob->spawnAnim();
124 delay();
125 }
126 }
127 }
128 }
129
130 TileEntity::tick();
131}

133void MobSpawnerTileEntity::fillExtraData(std::shared_ptr<Entity> entity) {
134 if (spawnData != NULL) {
135 CompoundTag* data = new CompoundTag();
136 entity->save(data);
137
138 std::vector<Tag*>* allTags = spawnData->getAllTags();
139 for (AUTO_VAR(it, allTags->begin()); it != allTags->end(); ++it) {
140 Tag* tag = *it;
141 data->put((wchar_t*)tag->getName().c_str(), tag->copy());
142 }
143 if (allTags != NULL) delete allTags;
144
145 entity->load(data);
146 }
