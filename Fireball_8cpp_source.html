116 // 4J-PB - Moved forward from 1.2.3
117 // if (!level->isClientSide && (owner == NULL || owner->removed))
118 if (!level->isClientSide) {
119 if ((owner != NULL && owner->removed) ||
120 !level->hasChunkAt((int)x, (int)y, (int)z)) {
121 app.DebugPrintf(
122 "Fireball removed - owner is null or removed is true for "
123 "owner\n");
124 remove();
125 return;
126 } else {
127 // 4J-PB - TU9 bug fix - fireballs can hit the edge of the world,
128 // and stay there
129 int minXZ = -(level->dimension->getXZSize() * 16) / 2;
130 int maxXZ = (level->dimension->getXZSize() * 16) / 2 - 1;
131
132 if ((x <= minXZ) || (x >= maxXZ) || (z <= minXZ) || (z >= maxXZ)) {
133 remove();

140 Entity::tick();
141
142 // app.DebugPrintf("Fireball x %d, y %d, z%d\n",(int)x,(int)y,(int)z);
143
144 if (shouldBurn()) setOnFire(1);
145
146 if (inGround) {
147 int tile = level->getTile(xTile, yTile, zTile);
148 if (tile == lastTile) {
149 life++;
150 if (life == SharedConstants::TICKS_PER_SECOND * 30) {
151 remove();
152 app.DebugPrintf("Fireball removed - life is 20*60\n");
153 }
154 return;
155 } else {
156 inGround = false;
157
158 xd *= random->nextFloat() * 0.2f;
159 yd *= random->nextFloat() * 0.2f;
160 zd *= random->nextFloat() * 0.2f;
161 life = 0;
162 flightTime = 0;
163 }
164 } else {
165 flightTime++;
166 }
167
168 MemSect(41);
169 Vec3* from = Vec3::newTemp(x, y, z);
170 Vec3* to = Vec3::newTemp(x + xd, y + yd, z + zd);
171 HitResult* res = level->clip(from, to);
172
173 from = Vec3::newTemp(x, y, z);
174 to = Vec3::newTemp(x + xd, y + yd, z + zd);
175 if (res != NULL) {
176 to = Vec3::newTemp(res->pos->x, res->pos->y, res->pos->z);
177 }
178 std::shared_ptr<Entity> hitEntity = nullptr;
179 std::vector<std::shared_ptr<Entity> >* objects = level->getEntities(
180 shared_from_this(), this->bb->expand(xd, yd, zd)->grow(1, 1, 1));
181 double nearest = 0;
182 AUTO_VAR(itEnd, objects->end());
183 for (AUTO_VAR(it, objects->begin()); it != itEnd; it++) {
184 std::shared_ptr<Entity> e = *it; // objects->at(i);
185 if (!e->isPickable() || (e->is(owner)))
186 continue; // 4J Stu - Never collide with the owner (Enderdragon) //
187 // && flightTime < 25)) continue;
188
189 float rr = 0.3f;
190 AABB* bb = e->bb->grow(rr, rr, rr);
191 HitResult* p = bb->clip(from, to);
192 if (p != NULL) {
193 double dd = from->distanceTo(p->pos);
194 if (dd < nearest || nearest == 0) {
195 hitEntity = e;
196 nearest = dd;
197 }
198 delete p;
199 }
200 }
201
202 if (hitEntity != NULL) {
203 delete res;
204 res = new HitResult(hitEntity);
205 }
206 MemSect(0);
207
208 if (res != NULL) {
209 onHit(res);
210 }
211 delete res;
212 x += xd;
213 y += yd;
214 z += zd;
215
216 double sd = sqrt(xd * xd + zd * zd);
217 yRot = (float)(atan2(xd, zd) * 180 / PI);
218 xRot = (float)(atan2(yd, sd) * 180 / PI);
219
220 while (xRot - xRotO < -180) xRotO -= 360;
221 while (xRot - xRotO >= 180) xRotO += 360;
222
223 while (yRot - yRotO < -180) yRotO -= 360;
224 while (yRot - yRotO >= 180) yRotO += 360;
225
226 xRot = xRotO + (xRot - xRotO) * 0.2f;
227 yRot = yRotO + (yRot - yRotO) * 0.2f;
228
229 float inertia = 0.95f;
230 if (isInWater()) {
231 for (int i = 0; i < 4; i++) {
232 float s = 1 / 4.0f;
233 level->addParticle(eParticleType_bubble, x - xd * s, y - yd * s,
234 z - zd * s, xd, yd, zd);
235 }
236 inertia = 0.80f;
237 }
238
239 xd += xPower;
240 yd += yPower;
241 zd += zPower;
242 xd *= inertia;
243 yd *= inertia;
244 zd *= inertia;
245
246 // 4J-PB - bug fix for the fireballs in a saved game - they are saved with
247 // no/very small velocity, so end up hanging around in the air
248 if (!level->isClientSide) {
249 if ((abs(xd) < 0.002) && (abs(yd) < 0.002) && (abs(zd) < 0.002)) {
250 xd = 0.0;
251 zd = 0.0;
252 yd = 0.0;
253 app.DebugPrintf("Removing a fireball with zero velocity\n");
254 remove();
255 }
256 }
257
258 level->addParticle(getTrailParticleType(), x, y + 0.5f, z, 0, 0.01, 0);
259
260 setPos(x, y, z);
