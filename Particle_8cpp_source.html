59 float dd = (float)(Mth::sqrt(xd * xd + yd * yd + zd * zd));

98void Particle::tick() {
99 xo = x;
100 yo = y;
101 zo = z;
102
103 if (age++ >= lifetime) remove();
104
105 yd -= 0.04 * gravity;
106 move(xd, yd, zd);
107 xd *= 0.98f;
108 yd *= 0.98f;
109 zd *= 0.98f;

113 zd *= 0.7f;
