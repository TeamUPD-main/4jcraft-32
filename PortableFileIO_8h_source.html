48inline BinaryReadResult ReadBinaryFile(const std::wstring& path, void* buffer,
49 std::size_t capacity) {
50 std::FILE* stream = OpenBinaryFileForRead(path);
51 if (stream == NULL) {
52 return {BinaryReadStatus::not_found, 0, 0};
53 }
54
55 if (!Seek(stream, 0, SEEK_END)) {
56 std::fclose(stream);
57 return {BinaryReadStatus::read_error, 0, 0};
58 }
59
60 const __int64 endPosition = Tell(stream);
61 if (endPosition < 0) {
62 std::fclose(stream);
63 return {BinaryReadStatus::read_error, 0, 0};
64 }
65
66 const std::size_t fileSize = static_cast<std::size_t>(endPosition);
67 if (fileSize > capacity) {
68 std::fclose(stream);
69 return {BinaryReadStatus::too_large, 0, fileSize};
70 }
71
72 if (!Seek(stream, 0, SEEK_SET)) {
73 std::fclose(stream);
74 return {BinaryReadStatus::read_error, 0, fileSize};
75 }
76
77 const std::size_t bytesRead = std::fread(buffer, 1, fileSize, stream);
78 const bool failed = std::ferror(stream) != 0;
79 std::fclose(stream);
80 if (failed || bytesRead != fileSize) {
81 return {BinaryReadStatus::read_error, bytesRead, fileSize};
82 }
83
84 return {BinaryReadStatus::ok, bytesRead, fileSize};
85}
