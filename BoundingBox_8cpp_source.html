21BoundingBox* BoundingBox::orientBox(int footX, int footY, int footZ, int offX,
22 int offY, int offZ, int width, int height,
23 int depth, int orientation) {
24 switch (orientation) {
25 default:
26 return new BoundingBox(footX + offX, footY + offY, footZ + offZ,
27 footX + width - 1 + offX,
28 footY + height - 1 + offY,
29 footZ + depth - 1 + offZ);
30 case Direction::NORTH:
31 // foot is at x0, y0, z1
32 return new BoundingBox(footX + offX, footY + offY,
33 footZ - depth + 1 + offZ,
34 footX + width - 1 + offX,
35 footY + height - 1 + offY, footZ + offZ);
36 case Direction::SOUTH:
37 // foot is at x0, y0, z0
38 return new BoundingBox(footX + offX, footY + offY, footZ + offZ,
39 footX + width - 1 + offX,
40 footY + height - 1 + offY,
41 footZ + depth - 1 + offZ);
42 case Direction::WEST:
43 // foot is at x1, y0, z0, but width and depth are flipped
44 return new BoundingBox(footX - depth + 1 + offZ, footY + offY,
45 footZ + offX, footX + offZ,
46 footY + height - 1 + offY,
47 footZ + width - 1 + offX);
48 case Direction::EAST:
49 // foot is at x0, y0, z0, but width and depth are flipped
50 return new BoundingBox(footX + offZ, footY + offY, footZ + offX,
51 footX + depth - 1 + offZ,
52 footY + height - 1 + offY,
53 footZ + width - 1 + offX);
54 }
55}
