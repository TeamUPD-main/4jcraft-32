9void ArchiveFile::_readHeader(DataInputStream* dis) {
10 int numberOfFiles = dis->readInt();
11
12 for (int i = 0; i < numberOfFiles; i++) {
13 MetaData* meta = new MetaData();
14 meta->filename = dis->readUTF();
15 meta->ptr = dis->readInt();
16 meta->filesize = dis->readInt();
17
18 // Filenames preceeded by an asterisk have been compressed.
19 if (meta->filename[0] == '*') {
20 meta->filename = meta->filename.substr(1);
21 meta->isCompressed = true;
22 } else
23 meta->isCompressed = false;
24
25 m_index.insert(
26 std::pair<std::wstring, PMetaData>(meta->filename, meta));
27 }
28}

89byteArray ArchiveFile::getFile(const std::wstring& filename) {
90 byteArray out;
91 AUTO_VAR(it, m_index.find(filename));
92
93 if (it == m_index.end()) {
94 app.DebugPrintf("Couldn't find file in archive\n");
95 app.DebugPrintf("Failed to find file '%ls' in archive\n",
96 filename.c_str());
97#ifndef _CONTENT_PACKAGE
98 __debugbreak();
99#endif
100 app.FatalLoadError();
101 } else {
102 PMetaData data = it->second;
103
104#if defined _XBOX_ONE || defined __ORBIS__ || defined _WINDOWS64
105 out = byteArray(data->filesize);
106
107 memcpy(out.data, m_cachedData + data->ptr, data->filesize);
108#else
109 const unsigned int fileSize = static_cast<unsigned int>(data->filesize);
110 std::uint8_t* pbData = new std::uint8_t[fileSize == 0 ? 1 : fileSize];
111 out = byteArray(pbData, fileSize);
112 const PortableFileIO::BinaryReadResult readResult =
113 PortableFileIO::ReadBinaryFileSegment(
114 m_sourcefile.getPath(), static_cast<std::size_t>(data->ptr),
115 out.data, static_cast<std::size_t>(data->filesize));
116
117 if (readResult.status != PortableFileIO::BinaryReadStatus::ok) {
118 app.DebugPrintf("Failed to read archive file segment\n");
119 app.FatalLoadError();
120 }
121#endif
122
123 // Compressed filenames are preceeded with an asterisk.
124 if (data->isCompressed && out.data != NULL) {
125 /* 4J-JEV:
126 * If a compressed file is accessed before compression object is
127 * initialized it will crash here (Compression::getCompression).
128 */
130
131 ByteArrayInputStream bais(out);
132 DataInputStream dis(&bais);
133 unsigned int decompressedSize = dis.readInt();
134 dis.close();
135
136 std::uint8_t* uncompressedBuffer =
137 new std::uint8_t[decompressedSize];
138 Compression::getCompression()->Decompress(
139 uncompressedBuffer, &decompressedSize, out.data + 4,
140 out.length - 4);
141
142 delete[] out.data;
143
144 out.data = uncompressedBuffer;
145 out.length = decompressedSize;
146 }
147
148 assert(out.data != NULL); // THERE IS NO FILE WITH THIS NAME!
149 }
150
151 return out;
152}
